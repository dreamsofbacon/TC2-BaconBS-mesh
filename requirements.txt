# Runtime dependencies for the BBS and its web admin.
#
# meshcore requires Python 3.10+, so it carries an environment marker: on an
# older host pip skips it instead of refusing the WHOLE file. Without the
# marker a Python 3.9 node gets NOTHING installed -- including paho-mqtt --
# and MQTT then fails with no obvious connection to the Python version.
# MeshCore support is imported lazily, so a node without it runs fine unless
# a MeshCore radio is actually configured.
meshtastic
meshcore>=2.3.8,<3; python_version >= "3.10"
paho-mqtt>=2.1,<3
pypubsub
flask

# Imported directly by config_init.py for serial port discovery. meshtastic
# happens to pull it in today, but relying on another package's dependency
# for something we import ourselves breaks the day that changes.
pyserial

# Ed25519 verification for fleet update instructions (fleet_update.py). This
# is the security boundary for remote updates, so it is a hard requirement
# rather than an optional extra: a node that cannot verify a signature must
# not fall back to trusting the message.
#
# Declared explicitly and NOT taken from meshcore's transitive pycryptodome,
# which is skipped entirely on Python 3.9 -- one of the two live nodes runs
# 3.9 on armv7l, so the transitive route is absent exactly where it is
# needed. Confirmed installable there from piwheels.
cryptography>=41

# Optional SSH-facing BBS service. The listener remains disabled unless the
# [ssh] section explicitly enables it. Releases which support Python 3.9 have
# a known SCP path-traversal issue, so old nodes skip SSH rather than install
# an affected version. The BBS itself continues to run there unchanged.
asyncssh>=2.23.1,<3; python_version >= "3.10"
